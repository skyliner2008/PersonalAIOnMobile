import { GoogleGenerativeAI } from '@google/generative-ai';
import { vectorStore } from './vectorStore.js';
import { knowledgeGraph } from './knowledgeGraph.js';
import { AutoTradingConfig, AnalysisSummary } from './types.js';
 
// Deprecated/removed Gemini model names -> current replacement
// Reference: https://ai.google.dev/gemini-api/docs/models
const DEPRECATED_MODELS: Record<string, string> = {
  'gemini-2.0-flash':           'gemini-2.5-flash',
  'gemini-2.0-flash-lite':      'gemini-2.5-flash-lite',
  'gemini-1.5-flash':           'gemini-2.5-flash',
  'gemini-1.5-flash-lite':      'gemini-2.5-flash-lite',
  'gemini-1.5-pro':             'gemini-2.5-flash',
  'gemini-pro':                 'gemini-2.5-flash',
  'gemini-1.0-pro':             'gemini-2.5-flash',
  'gemini-ultra':               'gemini-2.5-flash',
};
 
// Default generation model (stable, high token limit, low cost)
const DEFAULT_MODEL = 'gemini-2.5-flash-lite';
 
// Embedding model cascade: newest first.
// `embedding-001` has been fully deprecated (returns 404 as of 2026-04).
// `gemini-embedding-001` is the current recommended model; its output is
// Matryoshka-trained, so we can safely truncate to 768 dims and L2-normalize
// to match the 768-dim HNSW index in vectorStore.
const EMBEDDING_MODELS = [
  { model: 'gemini-embedding-001', apiVersion: 'v1beta' as const },
  { model: 'text-embedding-004',   apiVersion: 'v1beta' as const },
];
 
// Target dimensionality — MUST match vectorStore.DIMENSION (768).
const EMBEDDING_TARGET_DIMS = 768;
 
/**
 * Matryoshka-style downsize: take the first N components and L2-normalize.
 * This is the method Google recommends when reducing gemini-embedding-001
 * output below its native 3072 dims. Safe no-op when the input is already
 * <= target size (e.g. text-embedding-004 returns 768 natively).
 */
function fitDimensions(vec: number[], target: number): number[] {
  const sliced = vec.length > target ? vec.slice(0, target) : vec.slice();
  let sumSq = 0;
  for (const v of sliced) sumSq += v * v;
  const norm = Math.sqrt(sumSq);
  if (!isFinite(norm) || norm === 0) return sliced;
  for (let i = 0; i < sliced.length; i++) sliced[i] /= norm;
  return sliced;
}
 
function resolveModel(name: string | undefined): string {
  const m = (name || DEFAULT_MODEL).trim();
  const replacement = DEPRECATED_MODELS[m];
  if (replacement) {
    console.warn(`[Agent] Model "${m}" is deprecated -- using "${replacement}" instead.`);
    return replacement;
  }
  return m;
}
 
class AgentOrchestrator {
  private genAI: GoogleGenerativeAI | null = null;
 
  private init(apiKey: string) {
    if (!this.genAI && apiKey) {
      this.genAI = new GoogleGenerativeAI(apiKey);
    }
  }
 
  /**
   * 4-layer AI reasoning:
   * Layer 1 - Relational: config rules
   * Layer 2 - Vector:     similar past snapshots (HNSW)
   * Layer 3 - Graph:      strategy-symbol knowledge (SQLite)
   * Layer 4 - Reasoning:  Gemini LLM
   */
  public async reason(
    config: AutoTradingConfig,
    symbol: string,
    analyses: Record<string, AnalysisSummary>,
    historyContext: string,
    correlations: any[] = []
  ): Promise<string> {
    if (!config.apiKey) return 'No API Key provided for AI reasoning.';
    this.init(config.apiKey);
 
    const modelName = resolveModel(config.aiModel);
    const model = this.genAI!.getGenerativeModel({ model: modelName });
 
    // --- Layer 1: Relational ---
    const ruleContext = [
      'Current Strategy Rules:',
      `- Min Confluence: ${config.minConfluence}`,
      `- Risk Per Trade: ${config.risk.riskPerTradePct}%`,
      `- MTF Data Available: ${Object.entries(analyses).map(([tf, a]) => `${tf}(Bias:${a.bias},Conf:${a.confluence.toFixed(1)})`).join(' | ')}`,
      `- Preferred Strategies: ${config.preferredStrategies.length > 0 ? config.preferredStrategies.join(', ') : 'any'}`,
      `- Same-symbol positions allowed: ${config.adaptive?.allowSameSymbolPositions !== false}`,
      `- Over-limit defense allowed: ${config.adaptive?.allowOverLimitDefense !== false}`,
      `- Counter-hedge allowed: ${config.adaptive?.allowCounterHedge !== false}`,
      `- Scale-in recovery allowed: ${config.adaptive?.allowScaleInRecovery !== false}`,
      `- Hedge ratio range: ${config.adaptive?.hedgeRatioMin ?? 0.25}..${config.adaptive?.hedgeRatioMax ?? 1.0}`,
      `- Scale-in ratio range: ${config.adaptive?.scaleInRatioMin ?? 0.2}..${config.adaptive?.scaleInRatioMax ?? 0.75}`,
    ].join('\n');
 
    // --- Layer 2: Vector ---
    let vectorContext = '';
    try {
      const primary = analyses[config.timeframe || 'H1'] || Object.values(analyses)[0];
      const snapshotText = `Market: ${symbol}, Regime: ${primary.regime}, Bias: ${primary.bias}, Confluence: ${primary.confluence.toFixed(1)}, Fitness: ${primary.fitness.toFixed(1)}`;
      const queryEmbedding = await this.getEmbedding(config.apiKey, snapshotText);
      const isZeroVector = queryEmbedding.every(v => v === 0);
      if (!isZeroVector) {
        const similarSnapshots = vectorStore.search(queryEmbedding, 3, { symbol });
        if (similarSnapshots.length > 0) {
          const snippets = similarSnapshots.map((hit, i) => {
            const meta = hit.metadata;
            if (!meta) return null;
            return `  #${i + 1}: ${meta.symbol}/${meta.timeframe} [${meta.type}] at ${new Date(meta.timestamp).toISOString()} (distance=${hit.distance?.toFixed(3) ?? 'n/a'})`;
          }).filter(Boolean).join('\n');
          vectorContext = `[Vector Context]: ${similarSnapshots.length} similar past market snapshot(s) retrieved:\n${snippets}`;
        }
      } else {
        vectorContext = '[Vector Context]: Embedding unavailable (zero vector) -- skipping similarity search.';
      }
    } catch (vecErr) {
      console.error('[Agent] Vector retrieval failed:', vecErr);
    }
 
    // --- Layer 3: Graph ---
    let graphContext = `[Graph Context]: No structural knowledge available for ${symbol} yet.`;
    try {
      const symbolNodeId = `symbol_${symbol}`;
      const neighbors = knowledgeGraph.getContext(symbolNodeId, 1);
      if (neighbors.length > 0) {
        const lines = neighbors.map((n: any) => {
          const props = (() => { try { return JSON.parse(n.properties_json || '{}'); } catch { return {}; } })();
          const winRateStr = props.winRate !== undefined ? ` [winRate=${(props.winRate * 100).toFixed(1)}%]` : '';
          return `  - ${n.relation} -> ${n.label} (${n.type})${winRateStr}`;
        });
        graphContext = `[Graph Context]: Structural knowledge for ${symbol}:\n${lines.join('\n')}`;
      }
    } catch (graphErr) {
      console.error('[Agent] Graph retrieval failed:', graphErr);
    }
 
    // --- Layer 4: Reasoning ---
    const systemPrompt = config.agentPrompt || [
      'You are an AI Trading Agent with a strict operating playbook.',
      'You must analyze market regime, current exposure, unrealized PnL, and risk before deciding.',
      'If positions are already open, think in this order: protect capital, improve structure, then seek profit.',
      'You may recommend HOLD, REDUCE, CLOSE, HEDGE, SCALE_IN, BREAK_EVEN, or TRAIL when appropriate.',
      'Act like a professional portfolio trader: defend first, average only with trend support, hedge partially before full reversal, and close decisively when risk is no longer justified.',
      'When at position limit (indicated in CONTEXT), you must shift focus to harvesting profit or defending existing trades. Skip new entries.',
      'Prefer structured, repeatable reasoning over intuition.'
    ].join(' ');
 
    const prompt = [
      systemPrompt,
      '',
      'CONTEXT:',
      ruleContext,
      '',
      vectorContext,
      '',
      graphContext,
      '',
      `Historical Context: ${historyContext}`,
      '',
      'CROSS-ASSET CORRELATIONS (Phase 1 Intelligence):',
      correlations.length > 0 ? correlations.map(c => `- ${c.evidence}`).join('\n') : 'No correlation data available.',
      '',
      'MARKET DATA (Multi-Timeframe Analysis):',
      Object.entries(analyses).map(([tf, data]) => `[${tf}]: ${data.rationale}`).join('\n'),
      '',
      'TASK:',
      '1. Perform Multi-Timeframe (MTF) analysis: Use HTF (H4/D1) for trend, MTF (H1) for regime, and LTF (M5/M15) for precision/LQ.',
      '2. Analyze feasibility based on historical context and past performance.',
      '3. Decide whether to trade (Action: BUY/SELL/SKIP).',
      '4. Classify position management need (HOLD/REDUCE/CLOSE/HEDGE/SCALE_IN/BREAK_EVEN/TRAIL).',
      '5. Choose a professional sizing fraction for any hedge/scale-in/reduce action.',
      '6. INNOVATE: You are encouraged to invent, refine, and name your own trading techniques based on observed patterns. Update your "playbook" notes with these refinements.',
      '7. Provide a short English rationale and a concise THAI summary in "rationale_th" using headers like [ทิศทาง], [เหตุผล], [แผนการ].',
      '8. Provide a confidence score (0-100).',
      '9. If you need to close a specific position, specify its ticket number in "target_ticket".',
      '10. Suggest the most appropriate timeframe for the next LTF scan in "timeframe".',
      '',
      'Respond in JSON format ONLY (no other text outside JSON):',
      '{ "action": "BUY|SELL|SKIP", "management": "HOLD|REDUCE|CLOSE|HEDGE|SCALE_IN|BREAK_EVEN|TRAIL", "size_fraction": 0.35, "target_ticket": 123456, "timeframe": "M15", "rationale": "English rationale", "rationale_th": "สรุปภาษาไทยแบบหัวข้อ...", "confidence": 0, "adjustment": "...", "risk_note": "...", "playbook": "Self-evolved strategy notes" }',
    ].join('\n');
 
    try {
      const result = await model.generateContent(prompt);
      return result.response.text();
    } catch (error) {
      console.error('[Agent] Reasoning failed:', error);
      return JSON.stringify({ action: 'SKIP', rationale: `AI Error: ${String(error)}`, confidence: 0 });
    }
  }
 
  /** Get embeddings with model cascade fallback. Always returns a 768-dim vector. */
  public async getEmbedding(apiKey: string, text: string): Promise<number[]> {
    if (!apiKey) return new Array(EMBEDDING_TARGET_DIMS).fill(0);
    this.init(apiKey);
    let lastErr: unknown;
    for (const { model: modelName, apiVersion } of EMBEDDING_MODELS) {
      try {
        const model = this.genAI!.getGenerativeModel({ model: modelName }, { apiVersion });
        const result = await model.embedContent(text);
        const raw = result.embedding?.values ?? [];
        if (raw.length === 0) throw new Error(`Empty embedding from ${modelName}`);
        return fitDimensions(raw, EMBEDDING_TARGET_DIMS);
      } catch (err: any) {
        lastErr = err;
        // Only continue the cascade on "model not found / gone" style errors.
        // For quota/transient/network errors, fail fast so the outer caller
        // can skip the cycle rather than hammer every fallback.
        const status = err?.status;
        const isModelMissing = status === 404 || status === 400;
        if (!isModelMissing) {
          console.error(`[Agent] Embedding call to "${modelName}" failed (status=${status ?? 'n/a'}):`, err?.message ?? err);
          break;
        }
        console.warn(`[Agent] Embedding model "${modelName}" unavailable (${status}), trying next...`);
      }
    }
    console.error('[Agent] All embedding models failed. Using zero vector.', lastErr);
    return new Array(EMBEDDING_TARGET_DIMS).fill(0);
  }
}
 
export const agentOrchestrator = new AgentOrchestrator();
