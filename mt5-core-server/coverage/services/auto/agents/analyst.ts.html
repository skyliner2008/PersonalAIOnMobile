import { GoogleGenerativeAI } from '@google/generative-ai';
import { AutoTradingConfig, AnalysisSummary } from '../types.js';
import { AnalystReport } from './types.js';
import { vectorStore } from '../vectorStore.js';
import { knowledgeGraph } from '../knowledgeGraph.js';
 
export class AnalystAgent {
  private genAI: GoogleGenerativeAI | null = null;
 
  constructor(apiKey: string) {
    if (apiKey) {
      this.genAI = new GoogleGenerativeAI(apiKey);
    }
  }
 
  public async analyze(
    config: AutoTradingConfig,
    symbol: string,
    analyses: Record<string, AnalysisSummary>,
    correlations: any[] = []
  ): Promise<AnalystReport> {
    if (!this.genAI) throw new Error('AnalystAgent: API Key missing');
 
    const model = this.genAI.getGenerativeModel({ model: 'gemini-3.1-pro-preview' }, { apiVersion: 'v1beta' }); // Official ID from documentation
    
    const primary = analyses[config.timeframe || 'H1'] || Object.values(analyses)[0];
 
    // --- Phase 4: Episodic Memory Retrieval ---
    let pastLessonsContext = '';
    try {
        const query = `Symbol: ${symbol}, Regime: ${primary.regime}, Bias: ${primary.bias}`;
        const queryEmbedding = await this.getEmbedding(config.apiKey || '', query);
        const similarities = vectorStore.search(queryEmbedding, 10, { type: 'trade_lesson' }); // Increased to 10 for Tier 1
        if (similarities.length > 0) {
            pastLessonsContext = similarities.map(s => {
                const meta = s.metadata;
                if (!meta) return '';
                return `- ${meta.text || 'N/A'} (Outcome: ${meta.outcome || 'N/A'})`;
            }).filter(t => t !== '').join('\n');
        }
    } catch (err: any) {
        console.warn(`[AnalystAgent] Memory retrieval failed (likely 429), proceeding without memory:`, err?.message || err);
        pastLessonsContext = 'Memory unavailable due to API limits. Relying on real-time technicals.';
    }
 
    const prompt = `
You are a Senior Market Analyst Agent. 
Your task is to provide a deep narrative and directional bias for ${symbol} based on MTF analysis and correlations.
 
DATA:
- Watchlist MTF: ${Object.entries(analyses).map(([tf, a]) => `${tf}(Bias:${a.bias},Conf:${a.confluence.toFixed(1)},RSI:${a.rsi},EMA20:${a.ema20})`).join(' | ')}
- Technical Details (HTF H4): Regime=${analyses['H4']?.regime}, RSI=${analyses['H4']?.rsi}, SMA20=${analyses['H4']?.sma20}, SMA50=${analyses['H4']?.sma50}, EMA20=${analyses['H4']?.ema20}
- Technical Details (MTF H1): Regime=${analyses['H1']?.regime}, RSI=${analyses['H1']?.rsi}, SMA20=${analyses['H1']?.sma20}, SMA50=${analyses['H1']?.sma50}, EMA20=${analyses['H1']?.ema20}
- Technical Details (LTF M15): Regime=${analyses['M15']?.regime}, RSI=${analyses['M15']?.rsi}, EMA20=${analyses['M15']?.ema20}
- Technical Details (LTF M5): Regime=${analyses['M5']?.regime}, RSI=${analyses['M5']?.rsi}, EMA20=${analyses['M5']?.ema20}
- Smart Money Concepts (SMC): ${Object.entries(analyses).map(([tf, a]) => `${tf}: ${a.signals.filter(s => s.name.includes('fvg') || s.name.includes('order_block') || s.name.includes('market_structure')).map(s => s.evidence).join('; ')}`).join(' | ')}
- Correlations: ${correlations.map(c => c.evidence).join('\n')}
- Raw Rationale H4: ${analyses['H4']?.rationale || 'N/A'}
- Raw Rationale H1: ${analyses['H1']?.rationale || 'N/A'}
 
PAST LESSONS (EPISODIC MEMORY):
${pastLessonsContext || 'No relevant past lessons found.'}
 
TASK:
1. Synthesize the MTF data into a single cohesive technical narrative. Focus on M5/M15 bounces for scalping opportunities.
2. **TARGETING (XAUUSD)**: 1.0 point = $1.00. Always aim for a TP that is at least 8.0-20.0 points (e.g., if entry 4720, TP 4728+) away from entry.
3. **REVERSAL SCALPING**: If price is already at or past your first SMC target, DO NOT suggest a trade in that direction unless there is a clear secondary target significantly further away.
4. Identify high-confluence SMC zones (where M5/M15 zones overlap with H1/H4). These are your "5-Star" setups.
5. Target a quick profit of 8.0-15.0 points (Gold: $8-$15) using tight SL based on SMC invalidation.
6. **PRIORITIZE FVG FILLS**: Look for unfilled FVGs nearby (e.g., if price bounces at 4723, target the FVG at 4729-4734).
7. **COUNTER-TREND OK**: You are encouraged to propose counter-trend scalps if price is at a major SMC level with reversal signals.
  - Target RRR: Minimum 1.5:1. Preferred 2.0:1 or higher for institutional quality.
  - TP Distance: Ensure TP is at least 8.0 - 25.0 points away for XAUUSD to avoid "too close" broker errors.
  - SL Distance: Place SL behind high-probability structural levels. MINIMUM distance is 1.5 points (e.g., if entry 4720, SL 4718.5 for BUY). 
  - DO NOT suggest "micro-scalps" with SL/TP smaller than 1.5 points.
8. Mention specific indicators (RSI, SMA, EMA) and SMC elements (FVG, Order Blocks) to justify your bias.
9. Suggest the most fitting StrategyType.
10. Output a concise but highly technical THAI summary for the user, explaining the "Why" behind the technicals.
 
Respond in JSON format ONLY:
{
  "symbol": "SYMBOL",
  "bias": "BULL|BEAR|NEUTRAL",
  "regime": "TRENDING|RANGING|BREAKOUT|VOLATILE",
  "confluence": 0-100,
  "fitness": 0-100,
  "suggestedStrategy": "STRATEGY",
  "suggestedSL": 0.0,
  "suggestedTP": 0.0,
  "confidence": 0-100,
  "narrative": "Technical analysis summary in English",
  "narrative_th": "Technical analysis summary in Thai",
  "keyLevels": { "support": [], "resistance": [] },
  "correlations": []
}
`;
 
    try {
      const result = await model.generateContent(prompt);
      const text = result.response.text();
      const aiData = JSON.parse(text.match(/\{.*\}/s)?.[0] || '{}');
      
      return {
        symbol,
        regime: primary.regime,
        confluence: primary.confluence,
        fitness: primary.fitness,
        bias: aiData.bias || primary.bias,
        narrative: aiData.narrative || 'No narrative provided',
        narrative_th: aiData.narrative_th || 'ไม่มีคำอธิบาย',
        confidence: aiData.confidence || 50,
        suggestedStrategy: aiData.suggestedStrategy || primary.strategy,
        suggestedSL: aiData.suggestedSL || null,
        suggestedTP: aiData.suggestedTP || null,
        keyLevels: aiData.keyLevels || { support: [], resistance: [] },
        correlations: aiData.correlations || []
      } as AnalystReport;
    } catch (error) {
      console.error('[AnalystAgent] Analysis failed:', error);
      throw error;
    }
  }
 
  private async getEmbedding(apiKey: string, text: string): Promise<number[]> {
    const genAI = new GoogleGenerativeAI(apiKey);
    // Explicitly using gemini-embedding-2 as per Tier 1 list
    const model = genAI.getGenerativeModel({ model: 'gemini-embedding-2' }, { apiVersion: 'v1beta' });
    
    let retries = 3;
    let delay = 5000; // Start with 5s for Tier 1 stability
 
    while (retries > 0) {
      try {
        const result = await model.embedContent({
          content: { parts: [{ text }], role: 'user' }
        });
        return result.embedding.values.slice(0, 768);
      } catch (error: any) {
        const isRateLimit = error.status === 429 || error.message?.includes('429') || error.message?.includes('Resource exhausted');
        if (isRateLimit && retries > 1) {
          console.warn(`[AnalystAgent] Embedding 429 (Rate Limit) on 'gemini-embedding-2'. Retrying in ${delay/1000}s... (Attempts left: ${retries-1})`);
          await new Promise(r => setTimeout(r, delay));
          retries--;
          delay *= 2; 
        } else {
          throw error;
        }
      }
    }
    throw new Error('Failed to get embedding after retries');
  }
}
